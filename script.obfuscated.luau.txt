-- 3. VEHICLE DETECTION & VALIDATED SEATING SYSTEM (BYPASS KICK)
--------------------------------------------------------------------------------
local VehicleSystem = {}

-- ค้นหารถที่เป็นของผู้เล่นใน Workspace
function VehicleSystem.FindPlayerVehicle(character)
    local root = character:FindFirstChild("HumanoidRootPart")
    if not root then return nil, nil, nil end

    local foundVehicle = nil
    local foundSeat = nil
    local foundPrompt = nil
    local minDistance = math.huge

    local vehiclesFolder = Workspace:FindFirstChild("Vehicles") or Workspace
    
    for _, obj in ipairs(vehiclesFolder:GetChildren()) do
        if obj:IsA("Model") then
            local seat = obj:FindFirstChildWhichIsA("VehicleSeat", true) or obj:FindFirstChildWhichIsA("Seat", true)
            local prompt = obj:FindFirstChild("DrivePrompt", true) or obj:FindFirstChildWhichIsA("ProximityPrompt", true)
            
            if seat then
                local ownerAttr = obj:GetAttribute("Owner") or obj:GetAttribute("OwnerUserId") or obj:GetAttribute("Player")
                local isOwner = (ownerAttr == LocalPlayer.Name or ownerAttr == LocalPlayer.UserId)
                
                local dist = (seat.Position - root.Position).Magnitude
                if isOwner or dist < minDistance then
                    minDistance = dist
                    foundVehicle = obj
                    foundSeat = seat
                    foundPrompt = prompt
                    if isOwner then break end
                end
            end
        end
    end

    return foundVehicle, foundSeat, foundPrompt
end

-- ดึงรถ + กด DrivePrompt เข้าเบาะอย่างถูกต้อง 100% ไม่โดนคิก
function VehicleSystem.PrepareVehicleAndSeat(character)
    local humanoid = character:FindFirstChildOfClass("Humanoid")
    local root = character:FindFirstChild("HumanoidRootPart")

    -- ตรวจสอบว่านั่งอยู่บนรถเรียบร้อยแล้วหรือไม่
    if humanoid and humanoid.SeatPart then
        local seat = humanoid.SeatPart
        local model = seat:FindFirstAncestorOfClass("Model") or seat.Parent
        State.VehicleModel = model
        State.VehicleSeat = seat
    else
        -- ค้นหารถและดึงมาหาตัวผู้เล่น
        local vehicle, seat, prompt = VehicleSystem.FindPlayerVehicle(character)
        if vehicle and seat then
            State.VehicleModel = vehicle
            State.VehicleSeat = seat

            -- ดึงรถมาจอดตรงหน้าผู้เล่น
            if vehicle.PrimaryPart then
                vehicle:PivotTo(root.CFrame * CFrame.new(0, 2, -4))
            else
                seat.CFrame = root.CFrame * CFrame.new(0, 2, -4)
            end
            task.wait(0.15)

            -- เรียกกดปุ่ม DrivePrompt อัตโนมัติผ่าน fireproximityprompt (Bypass Validation Kick)
            if prompt then
                pcall(function()
                    root.CFrame = prompt.Parent:IsA("BasePart") and prompt.Parent.CFrame or seat.CFrame
                    task.wait(0.05)
                    if typeof(fireproximityprompt) == "function" then
                        fireproximityprompt(prompt)
                    else
                        -- Fallback หาก executor ไม่สนับสนุน fireproximityprompt
                        prompt:InputHoldBegin()
                        task.wait(0.05)
                        prompt:InputHoldEnd()
                    end
                end)
                task.wait(0.3)
            else
                -- Fallback นั่งเบาะกรณีไม่พบ Prompt
                pcall(function() seat:Sit(humanoid) end)
                task.wait(0.2)
            end
        end
    end

    -- คำนวณความสูงลอยเหนือพื้นป้องกันท้องรถ/ล้อจมดิน
    if State.VehicleModel then
        local success, _, size = pcall(function() return State.VehicleModel:GetBoundingBox() end)
        if success and size then
            State.VehicleHeightOffset = (size.Y / 2) + 0.8
        else
            State.VehicleHeightOffset = 3.0
        end
    else
        State.VehicleHeightOffset = 0.5
    end

    print(string.format("🚗 [VehicleSystem] Seat Validated & Prepared! Anti-Sinking Height Offset: %.2f studs", State.VehicleHeightOffset))
end

-- 4. HELPER & ENVIRONMENT SETUP

local Helper = {}

function Helper.GetIgnoreList(character)
    local ignore = { character }
    if State.VehicleModel then table.insert(ignore, State.VehicleModel) end
    if State.VisualFolder then table.insert(ignore, State.VisualFolder) end
    for _, part in ipairs(State.IgnoredParts) do table.insert(ignore, part) end
    return ignore
end

--------------------------------------------------------------------------------
-- 5. HEAVY MULTI-RAYCAST MATRIX SCANNER
--------------------------------------------------------------------------------
local RayScanner = {}

function RayScanner.ScanSurroundings(currentPos, targetDir, character)
    local ignoreList = Helper.GetIgnoreList(character)
    local params = RaycastParams.new()
    params.FilterType = Enum.RaycastFilterType.Exclude
    params.FilterDescendantsInstances = ignoreList
    params.IgnoreWater = true

    -- 1. Raycast ดิ่งลงพื้นเพื่อหา Y-Level พื้นจริง + บวกระยะลอยกันรถจม
    local groundRayOrigin = Vector3.new(currentPos.X, currentPos.Y + 25, currentPos.Z)
    local groundHit = Workspace:Raycast(groundRayOrigin, Vector3.new(0, -70, 0), params)
    local groundY = groundHit and (groundHit.Position.Y + State.VehicleHeightOffset) or (currentPos.Y + State.VehicleHeightOffset)

    -- 2. Forward Rays 3 ระดับความสูง
    local feetPos = Vector3.new(currentPos.X, groundY + 0.5, currentPos.Z)
    local waistPos = Vector3.new(currentPos.X, groundY + 3.5, currentPos.Z)
    local headPos = Vector3.new(currentPos.X, groundY + 7.0, currentPos.Z)

    local scanDist = CONFIG.RAYCAST_MATRIX.SCAN_DISTANCE
    local lowHit = Workspace:Raycast(feetPos, targetDir * scanDist, params)
    local midHit = Workspace:Raycast(waistPos, targetDir * scanDist, params)
    local headHit = Workspace:Raycast(headPos, targetDir * scanDist, params)

    local mainObstacle = lowHit or midHit or headHit

    -- กรณีไม่มีสิ่งกีดขวางด้านหน้า
    if not mainObstacle or not mainObstacle.Instance or not mainObstacle.Instance.CanCollide then
        return {
            Type = "CLEAR",
            Position = Vector3.new(currentPos.X, groundY, currentPos.Z)
        }
    end

    -- คำนวณความสูงของสิ่งกีดขวาง
    local hitPart = mainObstacle.Instance
    local partTopY = hitPart.Position.Y + (hitPart.Size.Y / 2)
    local heightDiff = partTopY - groundY

    -- 3. กรณีเจอกำแพงเตี้ย (Vaultable): ยกรถและเส้นลอยข้าม
    if heightDiff > 0 and heightDiff <= CONFIG.RAYCAST_MATRIX.MAX_VAULT_HEIGHT then
        local elevatedY = partTopY + CONFIG.RAYCAST_MATRIX.SAFETY_CLEARANCE + State.VehicleHeightOffset
        return {
            Type = "VAULT",
            Position = Vector3.new(currentPos.X, elevatedY, currentPos.Z)
        }
    end

    -- 4. กรณีเจอกำแพงสูง (Tall Wall): ยิง Raycast หลายมุมเพื่อคำนวณทางอ้อม
    local bestDetourDir = nil
    local maxClearDistance = 0

    for _, angleDeg in ipairs(CONFIG.RAYCAST_MATRIX.SIDE_ANGLES) do
        local radRight = math.rad(angleDeg)
        local radLeft = math.rad(-angleDeg)

        local dirRight = (CFrame.Angles(0, radRight, 0) * CFrame.new(Vector3.zero, targetDir)).LookVector
        local dirLeft = (CFrame.Angles(0, radLeft, 0) * CFrame.new(Vector3.zero, targetDir)).LookVector

        local hitRight = Workspace:Raycast(waistPos, dirRight * (scanDist + 12), params)
        local hitLeft = Workspace:Raycast(waistPos, dirLeft * (scanDist + 12), params)

        local distRight = hitRight and hitRight.Distance or (scanDist + 12)
        local distLeft = hitLeft and hitLeft.Distance or (scanDist + 12)

        if distRight > maxClearDistance then
            maxClearDistance = distRight
            bestDetourDir = dirRight
        end

        if distLeft > maxClearDistance then
            maxClearDistance = distLeft
            bestDetourDir = dirLeft
        end
    end

    if bestDetourDir then
        local detourPos = currentPos + (bestDetourDir * CONFIG.RAYCAST_MATRIX.WALL_DETOUR_MARGIN)
        local detourGroundHit = Workspace:Raycast(Vector3.new(detourPos.X, detourPos.Y + 25, detourPos.Z), Vector3.new(0, -70, 0), params)
        local finalDetourY = detourGroundHit and (detourGroundHit.Position.Y + State.VehicleHeightOffset) or (detourPos.Y + State.VehicleHeightOffset)
        
        return {
            Type = "DETOUR",
            Position = Vector3.new(detourPos.X, finalDetourY, detourPos.Z)
        }
    end

    return {
        Type = "CLEAR",
        Position = Vector3.new(currentPos.X, groundY, currentPos.Z)
    }
end

--------------------------------------------------------------------------------
-- 6. VISUAL PATH RENDERER (BEAM & NEON NODES)
--------------------------------------------------------------------------------
local VisualRenderer = {}

function VisualRenderer.Cleanup()
    if State.VisualFolder then
        State.VisualFolder:Destroy()
        State.VisualFolder = nil
    end
    local existing = Workspace:FindFirstChild(CONFIG.VISUALS.FOLDER_NAME)
    if existing then existing:Destroy() end
end

function VisualRenderer.RenderPath(waypoints)
    VisualRenderer.Cleanup()
    
    local folder = Instance.new("Folder")
    folder.Name = CONFIG.VISUALS.FOLDER_NAME
    folder.Parent = Workspace
    State.VisualFolder = folder
    
    local totalPoints = #waypoints
    if totalPoints < 2 then return end

    for i = 1, totalPoints - 1 do
        local pA = waypoints[i]
        local pB = waypoints[i + 1]
        
        local alphaA = (i - 1) / (totalPoints - 1)
        local alphaB = i / (totalPoints - 1)
        local colorA = CONFIG.VISUALS.COLOR_START:Lerp(CONFIG.VISUALS.COLOR_END, alphaA)
        local colorB = CONFIG.VISUALS.COLOR_START:Lerp(CONFIG.VISUALS.COLOR_END, alphaB)

        local partA = Instance.new("Part")
        partA.Name = "Node_" .. i
        partA.Size = Vector3.new(0.1, 0.1, 0.1)
        partA.Position = pA
        partA.Anchored = true
        partA.CanCollide = false
        partA.Transparency = 1
        partA.Parent = folder

        local partB = Instance.new("Part")
        partB.Name = "Node_" .. (i + 1)
        partB.Size = Vector3.new(0.1, 0.1, 0.1)
        partB.Position = pB
        partB.Anchored = true
        partB.CanCollide = false
        partB.Transparency = 1
        partB.Parent = folder

        local attA = Instance.new("Attachment", partA)
        local attB = Instance.new("Attachment", partB)

        local beam = Instance.new("Beam")
        beam.Name = "PathBeam_" .. i
        beam.Attachment0 = attA
        beam.Attachment1 = attB
        beam.Color = ColorSequence.new(colorA, colorB)
        beam.Width0 = CONFIG.VISUALS.BEAM_WIDTH
        beam.Width1 = CONFIG.VISUALS.BEAM_WIDTH
        beam.Transparency = CONFIG.VISUALS.BEAM_TRANSPARENCY
        beam.FaceCamera = true
        beam.TextureSpeed = CONFIG.VISUALS.BEAM_SPEED
        beam.Parent = folder

        if CONFIG.VISUALS.SHOW_NODE_MARKERS then
            local marker = Instance.new("Part")
            marker.Name = "Marker_" .. i
            marker.Shape = Enum.PartType.Cylinder
            marker.Size = Vector3.new(0.05, 0.6, 0.6)
            marker.CFrame = CFrame.new(pA) * CFrame.Angles(0, 0, math.rad(90))
            marker.Anchored = true
            marker.CanCollide = false
            marker.Material = Enum.Material.Neon
            marker.Color = colorA
            marker.Transparency = 0.25
            marker.Parent = folder
        end
    end
end

-- 7. SMART PATH GENERATION ENGINE
--------------------------------------------------------------------------------
local PathEngine = {}

function PathEngine.GenerateUltraSmartPath(startPos, destinationPos, character)
    print("[PathEngine] Calculating Smart Detour Path & Multi-Raycast Matrix...")

    local path = PathfindingService:CreatePath(CONFIG.PATH_PARAMS)
    local success, _ = pcall(function()
        path:ComputeAsync(startPos, destinationPos)
    end)

    local rawPoints = {}
    if success and path.Status == Enum.PathStatus.Success then
        for _, wp in ipairs(path:GetWaypoints()) do
            table.insert(rawPoints, wp.Position)
        end
    else
        table.insert(rawPoints, startPos)
        local dir = (destinationPos - startPos).Unit
        local dist = (destinationPos - startPos).Magnitude
        local steps = math.ceil(dist / CONFIG.PATH_PARAMS.WaypointSpacing)
        for i = 1, steps do
            table.insert(rawPoints, startPos + (dir * (i * CONFIG.PATH_PARAMS.WaypointSpacing)))
        end
        table.insert(rawPoints, destinationPos)
    end

    local processedWaypoints = {}
    for i = 1, #rawPoints do
        local curr = rawPoints[i]
        local nextPos = rawPoints[i + 1] or destinationPos
        local dir = (nextPos - curr).Magnitude > 0 and (nextPos - curr).Unit or Vector3.new(0, 0, -1)
        
        local scanResult = RayScanner.ScanSurroundings(curr, dir, character)
        
        if scanResult.Type == "DETOUR" then
            table.insert(processedWaypoints, scanResult.Position)
        end
        
        table.insert(processedWaypoints, scanResult.Position)
    end

    return processedWaypoints
end

-- 8. DIRECT FUNCTION & SIGNAL FIRING APPLYJOB CLICK ENGINE
--------------------------------------------------------------------------------
local function TriggerApplyJobUIClick()
    task.spawn(function()
        task.wait(0.2)
        print("[ApplyJob] Executing Direct Function & Signal Firing UI Click...")
        
        pcall(function()
            local playerGui = LocalPlayer:WaitForChild("PlayerGui", 5)
            if not playerGui then return end
            
            local jobApp = playerGui:WaitForChild("JobApplication", 5)
            if not jobApp then return end
            
            local jobFrame = jobApp:WaitForChild("JobApplicationFrame", 5)
            if not jobFrame then return end
            
            local applyButton = jobFrame:WaitForChild("ApplyJob", 5)
            if applyButton then
                if typeof(getconnections) == "function" then
                    for _, conn in ipairs(getconnections(applyButton.MouseButton1Click)) do
                        pcall(function() conn:Fire() end)
                    end
                    for _, conn in ipairs(getconnections(applyButton.Activated)) do
                        pcall(function() conn:Fire() end)
                    end
                end

                if typeof(firesignal) == "function" then
                    pcall(function() firesignal(applyButton.MouseButton1Click) end)
                    pcall(function() firesignal(applyButton.Activated) end)
                end

                local pos = applyButton.AbsolutePosition
                local size = applyButton.AbsoluteSize
                local centerX = pos.X + (size.X / 2)
                local centerY = pos.Y + (size.Y / 2) + 36

                VirtualInputManager:SendMouseButtonEvent(centerX, centerY, 0, true, game, 0)
                task.wait(0.04)
                VirtualInputManager:SendMouseButtonEvent(centerX, centerY, 0, false, game, 0)

                pcall(function()
                    GuiService.SelectedObject = applyButton
                    VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.Return, false, game)
                    task.wait(0.04)
                    VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.Return, false, game)
                    GuiService.SelectedObject = nil
                end)

                print("✅ จำลองการกดเมาส์ลงบนปุ่ม ApplyJob สำเร็จ!")
            end
        end)
    end)
end

-- 9. VEHICLE & CHARACTER TWEEN CONTROLLER (SPEED 45)
--------------------------------------------------------------------------------
local NavigationController = {}

function NavigationController.Stop()
    State.IsNavigating = false
    if State.CurrentTween then
        State.CurrentTween:Cancel()
        State.CurrentTween = nil
    end
    VisualRenderer.Cleanup()
end

function NavigationController.StartNavigation(destinationPos)
    local character = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
    local root = character:WaitForChild("HumanoidRootPart")

    -- 1. ตรวจสอบและเตรียมรถ + กดเปิด DrivePrompt ผ่าน fireproximityprompt (Bypass Validation Kick)
    VehicleSystem.PrepareVehicleAndSeat(character)

    local targetPart = root
    if State.VehicleModel then
        targetPart = State.VehicleModel.PrimaryPart or State.VehicleSeat or root
    end

    NavigationController.Stop()
    State.IsNavigating = true

    local startPos = targetPart.Position
    local waypoints = PathEngine.GenerateUltraSmartPath(startPos, destinationPos, character)

    if not waypoints or #waypoints < 2 then
        warn("[NavigationController] Failed to generate path!")
        return false
    end

    -- วาดเส้นทาง Beam ลอยเหนือพื้นอย่างสวยงาม
    VisualRenderer.RenderPath(waypoints)
    print(string.format("[NavigationController] Path Rendered. Total Waypoints: %d", #waypoints))

    -- เคลื่อนที่ยานพาหนะและตัวละครไปด้วยกันด้วย Tween (Speed = 45 studs/s)
    task.spawn(function()
        for i = 1, #waypoints - 1 do
            if not State.IsNavigating then break end

            local currentPos = waypoints[i]
            local nextPos = waypoints[i + 1]
            local segmentDistance = (nextPos - currentPos).Magnitude

            if segmentDistance > 0.05 then
                local tweenTime = segmentDistance / CONFIG.TWEEN_SPEED
                local targetCFrame = CFrame.lookAt(nextPos, nextPos + (nextPos - currentPos).Unit)

                local tweenInfo = TweenInfo.new(
                    tweenTime,
                    Enum.EasingStyle.Linear,
                    Enum.EasingDirection.Out
                )

                if State.VehicleModel then
                    local primary = State.VehicleModel.PrimaryPart or targetPart
                    State.CurrentTween = TweenService:Create(primary, tweenInfo, { CFrame = targetCFrame })
                    State.CurrentTween:Play()
                    
                    task.spawn(function()
                        local elapsed = 0
                        local initialCFrame = State.VehicleModel:GetPivot()
                        while elapsed < tweenTime and State.IsNavigating do
                            local dt = task.wait()
                            elapsed = elapsed + dt
                            local alpha = math.clamp(elapsed / tweenTime, 0, 1)
                            State.VehicleModel:PivotTo(initialCFrame:Lerp(targetCFrame, alpha))
                        end
                    end)
                else
                    State.CurrentTween = TweenService:Create(targetPart, tweenInfo, { CFrame = targetCFrame })
                    State.CurrentTween:Play()
                end

                task.wait(tweenTime)
            end
        end

        -- เมื่อเดินทางถึงพิกัดเป้าหมาย
        if State.IsNavigating then
            print("[NavigationController] Destination Reached Successfully!")
            NavigationController.Stop()
            TriggerApplyJobUIClick()
        end
    end)

    return true
end
