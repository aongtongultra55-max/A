local FriendsList = {}

local function isFriend(player)
    if not player then return false end
    if FriendsList[player.Name] or FriendsList[player.DisplayName] then
        return true
    end
    return false
end

local function getPlayerNames()
    local list = {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer then
            table.insert(list, p.Name)
        end
    end
    table.sort(list)
    return list
end

local function getPlayerCar(player)
    if not player then return nil end
    local char = player.Character
    if not char then return nil end

    local hum = char:FindFirstChildOfClass("Humanoid")
    if hum and hum.SeatPart then
        local seat = hum.SeatPart
        local veh = seat:FindFirstAncestorOfClass("Model")
        if veh and (veh.Parent == Workspace:FindFirstChild("Vehicles") or veh:IsDescendantOf(Workspace)) then
            return veh
        end
        return seat.Parent
    end

    local vehicles = Workspace:FindFirstChild("Vehicles")
    if vehicles then
        for _, v in ipairs(vehicles:GetChildren()) do
            if char:IsDescendantOf(v) then
                return v
            end
        end
    end

    return nil
end

local function isLocking()
    if IsMobile then
        if Camera:GetAttribute("QLockMouseRobloxCamera") == true then
            return true
        end
        if UserInputService.MouseBehavior == Enum.MouseBehavior.LockCurrentPosition 
           or UserInputService.MouseBehavior == Enum.MouseBehavior.LockCenter then
            return true
        end
        local char = LocalPlayer.Character
        if char then
            local hum = char:FindFirstChildOfClass("Humanoid")
            if hum then
                for _, track in ipairs(hum:GetPlayingAnimationTracks()) do
                    local anim = track.Animation
                    if anim and (anim.AnimationId:find("97952042379271") or anim.Name:lower():find("ads") or anim.Name:lower():find("aim")) then
                        return true
                    end
                end
            end
        end
        return false
    else
        local selectedKey = KeyMap[Settings.AimKey]
        if selectedKey then
            if typeof(selectedKey) == "EnumItem" then
                if selectedKey.EnumType == Enum.UserInputType then
                    if UserInputService:IsMouseButtonPressed(selectedKey) then
                        return true
                    end
                elseif selectedKey.EnumType == Enum.KeyCode then
                    if UserInputService:IsKeyDown(selectedKey) then
                        return true
                    end
                end
            end
        end
        return false
    end
end

local function setupVehicle(veh)
    if veh and veh:IsA("Model") then
        veh:SetAttribute("IgnoreDisableTools", true)
    end
end

local vehiclesFolder = Workspace:FindFirstChild("Vehicles")
if vehiclesFolder then
    for _, v in ipairs(vehiclesFolder:GetChildren()) do
        setupVehicle(v)
    end
    vehiclesFolder.ChildAdded:Connect(setupVehicle)
end

local function applyNoRecoil(tool)
    if not tool or not tool:IsA("Tool") then return end

    if tool:GetAttribute("Recoil") ~= nil then
        if Settings.NoRecoil then
            tool:SetAttribute("Recoil", 0)
        end
    end

    tool:SetAttribute("DisableInVehicle", false)
    tool:SetAttribute("Disabled", false)

    if not tool:FindFirstChild("ToolTweaker") then
        local watcher = Instance.new("BoolValue")
        watcher.Name = "ToolTweaker"
        watcher.Parent = tool

        tool:GetAttributeChangedSignal("Recoil"):Connect(function()
            if Settings.NoRecoil and tool:GetAttribute("Recoil") ~= 0 then
                tool:SetAttribute("Recoil", 0)
            end
        end)

        tool:GetAttributeChangedSignal("Disabled"):Connect(function()
            if tool:GetAttribute("Disabled") == true then
                tool:SetAttribute("Disabled", false)
            end
        end)

        tool:GetAttributeChangedSignal("DisableInVehicle"):Connect(function()
            if tool:GetAttribute("DisableInVehicle") == true then
                tool:SetAttribute("DisableInVehicle", false)
            end
        end)
    end
end

local function scanAllTools()
    local myCar = getPlayerCar(LocalPlayer)
    if myCar and typeof(myCar) == "Instance" then
        myCar:SetAttribute("IgnoreDisableTools", true)
    end

    if LocalPlayer.Character then
        for _, item in ipairs(LocalPlayer.Character:GetChildren()) do
            applyNoRecoil(item)
        end
    end

    local backpack = LocalPlayer:FindFirstChildOfClass("Backpack")
    if backpack then
        for _, item in ipairs(backpack:GetChildren()) do
            applyNoRecoil(item)
        end
    end
end

local function setupInventory()
    local function onChar(char)
        local hum = char:WaitForChild("Humanoid", 5)
        if hum then
            hum:GetPropertyChangedSignal("SeatPart"):Connect(function()
                if hum.SeatPart then
                    local seat = hum.SeatPart
                    local veh = seat:FindFirstAncestorOfClass("Model")
                    if veh then
                        veh:SetAttribute("IgnoreDisableTools", true)
                    end
                    if seat.Parent and seat.Parent:IsA("Model") then
                        seat.Parent:SetAttribute("IgnoreDisableTools", true)
                    end
                end
                task.wait(0.05)
                scanAllTools()
            end)
        end

        char.ChildAdded:Connect(function(child)
            task.wait(0.05)
            applyNoRecoil(child)
        end)
        scanAllTools()
    end

    if LocalPlayer.Character then
        onChar(LocalPlayer.Character)
    end
    LocalPlayer.CharacterAdded:Connect(onChar)

    local backpack = LocalPlayer:WaitForChild("Backpack", 5)
    if backpack then
        backpack.ChildAdded:Connect(function(child)
            task.wait(0.05)
            applyNoRecoil(child)
        end)
    end
end

local function isBehindWall(targetPart, targetCharacter, targetPlayer)
    if not targetPart or not targetCharacter then return false end

    local origin = Camera.CFrame.Position
    local direction = targetPart.Position - origin

    local raycastParams = RaycastParams.new()
    local filterType = Enum.RaycastFilterType.Blacklist
    pcall(function()
        filterType = Enum.RaycastFilterType.Exclude
    end)
    raycastParams.FilterType = filterType

    local ignoreList = { Camera }
    if LocalPlayer.Character then
        table.insert(ignoreList, LocalPlayer.Character)
    end

    local myCar = getPlayerCar(LocalPlayer)
    if myCar and typeof(myCar) == "Instance" then
        table.insert(ignoreList, myCar)
    end

    table.insert(ignoreList, targetCharacter)

    local targetCar = getPlayerCar(targetPlayer)
    if targetCar and typeof(targetCar) == "Instance" then
        table.insert(ignoreList, targetCar)
    end

    raycastParams.FilterDescendantsInstances = ignoreList
    raycastParams.IgnoreWater = true

    local maxTries = 5
    local currentOrigin = origin
    local currentDir = direction

    while maxTries > 0 do
        maxTries = maxTries - 1
        local result = Workspace:Raycast(currentOrigin, currentDir, raycastParams)
        if not result then
            return false
        end

        local hitInstance = result.Instance
        if not hitInstance then
            return false
        end

        if hitInstance:IsDescendantOf(targetCharacter) or (targetCar and hitInstance:IsDescendantOf(targetCar)) then
            return false
        end

        if not hitInstance.CanCollide or hitInstance.Transparency >= 0.8 or hitInstance:IsA("Accessory") or hitInstance:IsA("Tool") then
            table.insert(ignoreList, hitInstance)
            raycastParams.FilterDescendantsInstances = ignoreList
            currentOrigin = result.Position + direction.Unit * 0.1
            currentDir = targetPart.Position - currentOrigin
            if currentDir.Magnitude <= 0.2 then
                return false
            end
        else
            return true
        end
    end

    return false
end

local function getTargetPart(character, partType)
    if not character then return nil end

    local resolvedType = partType
    if resolvedType == "SAFE" then
        local now = os.clock()
        if now - lastSafeSwapTime > 0.35 then
            lastSafeSwapTime = now
            if math.random(1, 2) == 1 then
                currentSafeModePart = "HEAD"
            else
                currentSafeModePart = "BODY"
            end
        end
        resolvedType = currentSafeModePart
    end

    if resolvedType == "HEAD" then
        return character:FindFirstChild("Head") or character:FindFirstChild("HumanoidRootPart")
    elseif resolvedType == "BODY" then
        return character:FindFirstChild("HumanoidRootPart")
            or character:FindFirstChild("UpperTorso")
            or character:FindFirstChild("Torso")
            or character:FindFirstChild("Head")
    end
    return character:FindFirstChild("Head")
end

local function isTargetValid(player, part)
    if not player or not player.Character or not part or not part.Parent then return false end
    if isFriend(player) then return false end

    local char = player.Character
    local hum = char:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then return false end
    if char:FindFirstChildOfClass("ForceField") then return false end

    if LocalPlayer.Team and player.Team and LocalPlayer.Team == player.Team and #Players:GetPlayers() > 2 then
        return false
    end

    local screenPos, onScreen = Camera:WorldToViewportPoint(part.Position)
    if not onScreen or screenPos.Z <= 0 then return false end

    local screenCenter = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
    local screenPoint = Vector2.new(screenPos.X, screenPos.Y)
    local distance = (screenPoint - screenCenter).Magnitude

    if distance > Settings.FOV * 1.25 then return false end

    if Settings.WallCheck then
        if isBehindWall(part, char, player) then
            return false
        end
    end

    return true
end

local function getClosestTarget()
    local screenCenter = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)
    local bestPlayer = nil
    local bestPart = nil
    local shortestDistance = Settings.FOV

    for _, player in ipairs(Players:GetPlayers()) do
        if player ~= LocalPlayer and player.Character and not isFriend(player) then
            local char = player.Character
            local hum = char:FindFirstChildOfClass("Humanoid")

            if hum and hum.Health > 0 and not char:FindFirstChildOfClass("ForceField") then
                local isTeam = LocalPlayer.Team and player.Team and LocalPlayer.Team == player.Team and #Players:GetPlayers() > 2
                if not isTeam then
                    local targetPart = getTargetPart(char, Settings.TargetPart)
                    if targetPart then
                        local screenPos, onScreen = Camera:WorldToViewportPoint(targetPart.Position)

                        if onScreen and screenPos.Z > 0 then
                            local screenPoint = Vector2.new(screenPos.X, screenPos.Y)
                            local distance = (screenPoint - screenCenter).Magnitude

                            if distance <= shortestDistance then
                                local canTarget = true
                                if Settings.WallCheck then
                                    if isBehindWall(targetPart, char, player) then
                                        canTarget = false
                                    end
                                end

                                if canTarget then
                                    shortestDistance = distance
                                    bestPlayer = player
                                    bestPart = targetPart
                                end
                            end
                        end
                    end
                end
            end
        end
    end

    return bestPlayer, bestPart
end

local FOVCircle = Drawing.new("Circle")
FOVCircle.Thickness = 1.5
FOVCircle.NumSides = 64
FOVCircle.Radius = Settings.FOV
FOVCircle.Filled = false
FOVCircle.Color = Settings.FOVColor
FOVCircle.Transparency = 1
FOVCircle.Visible = false

RunService.RenderStepped:Connect(function()
    local screenCenter = Vector2.new(Camera.ViewportSize.X / 2, Camera.ViewportSize.Y / 2)

    if Settings.SilentAim and Settings.ShowFOV then
        FOVCircle.Position = screenCenter
        FOVCircle.Radius = Settings.FOV
        FOVCircle.Visible = true
    else
        FOVCircle.Visible = false
    end

    if Settings.SilentAim and isLocking() then
        if CurrentTarget and CurrentTarget.Character and CurrentTargetPart and isTargetValid(CurrentTarget, CurrentTargetPart) then
            if Settings.TargetPart == "SAFE" then
                CurrentTargetPart = getTargetPart(CurrentTarget.Character, "SAFE")
            end
            Camera.CFrame = CFrame.lookAt(Camera.CFrame.Position, CurrentTargetPart.Position, Vector3.new(0, 1, 0))
        else
            local foundPlayer, foundPart = getClosestTarget()
            if foundPlayer and foundPart then
                CurrentTarget = foundPlayer
                CurrentTargetPart = foundPart
                Camera.CFrame = CFrame.lookAt(Camera.CFrame.Position, foundPart.Position, Vector3.new(0, 1, 0))
            else
                CurrentTarget = nil
                CurrentTargetPart = nil
            end
        end
    else
        CurrentTarget = nil
        CurrentTargetPart = nil
    end
end)

local function createFloatingButton()
    local parentGui
    if typeof(gethui) == "function" then
        parentGui = gethui()
    elseif CoreGui:FindFirstChild("RobloxGui") then
        parentGui = CoreGui
    else
        parentGui = LocalPlayer:WaitForChild("PlayerGui")
    end

    if parentGui:FindFirstChild("CombatHubFloatingToggle") then
        parentGui.CombatHubFloatingToggle:Destroy()
    end

    local ScreenGui = Instance.new("ScreenGui")
    ScreenGui.Name = "CombatHubFloatingToggle"
    ScreenGui.ResetOnSpawn = false
    ScreenGui.ZIndexBehavior = Enum.ZIndexBehavior.Sibling
    ScreenGui.Parent = parentGui

    local ToggleBtn = Instance.new("TextButton")
    ToggleBtn.Name = "ToggleBtn"
    ToggleBtn.Size = UDim2.new(0, 50, 0, 50)
    ToggleBtn.Position = UDim2.new(0, 20, 0.5, -25)
    ToggleBtn.BackgroundColor3 = Color3.fromRGB(25, 25, 25)
    ToggleBtn.Text = "🎯"
    ToggleBtn.TextSize = 22
    ToggleBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
    ToggleBtn.AutoButtonColor = true
    ToggleBtn.Active = true
    ToggleBtn.Parent = ScreenGui

    local UICorner = Instance.new("UICorner")
    UICorner.CornerRadius = UDim.new(1, 0)
    UICorner.Parent = ToggleBtn

    local UIStroke = Instance.new("UIStroke")
    UIStroke.Thickness = 2
    UIStroke.Color = Color3.fromRGB(16, 197, 80)
    UIStroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    UIStroke.Parent = ToggleBtn

    local dragging = false
    local dragInput, dragStart, startPos

    ToggleBtn.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
            dragging = true
            dragStart = input.Position
            startPos = ToggleBtn.Position

            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    dragging = false
                end
            end)
        end
    end)

    ToggleBtn.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging then
            local delta = input.Position - dragStart
            ToggleBtn.Position = UDim2.new(
                startPos.X.Scale,
                startPos.X.Offset + delta.X,
                startPos.Y.Scale,
                startPos.Y.Offset + delta.Y
            )
        end
    end)

    ToggleBtn.MouseButton1Click:Connect(function()
        Window:Toggle()
    end)
end
createFloatingButton()
